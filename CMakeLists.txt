cmake_minimum_required(VERSION 3.24)
project(VeilDesk VERSION 0.2.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Requires Windows and MSVC (Visual Studio 2022).")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use an x64 generator: -A x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
add_library(VeilCommon STATIC src/common/Logger.cpp src/common/ProgressWatchdog.cpp src/common/SystemInfo.cpp src/common/PreciseTimer.cpp)
target_include_directories(VeilCommon PUBLIC src)
add_library(VeilBroker STATIC src/broker/AuthorizationEnvelope.cpp src/broker/AuthorizationClient.cpp src/broker/Channel.cpp src/broker/ProcessIdentity.cpp src/broker/Protocol.cpp src/broker/SharedFrame.cpp src/broker/SharedFrameProducer.cpp)
target_link_libraries(VeilBroker PUBLIC VeilCommon d3d11 dxgi d3dcompiler advapi32 bcrypt)
add_library(VeilFrameReadback STATIC src/broker/FrameReadback.cpp)
target_link_libraries(VeilFrameReadback PUBLIC VeilBroker)
add_executable(SecureViewer WIN32 src/main.cpp src/window/SecureWindow.cpp src/graphics/D3D11Renderer.cpp src/security/DisplayProtection.cpp src/app.rc)
target_link_libraries(SecureViewer PRIVATE VeilBroker user32 shell32 d3d11 dxgi d2d1 dwrite dwmapi)
target_link_options(SecureViewer PRIVATE /MANIFEST:NO)
add_executable(CaptureTest src/capture/main.cpp src/capture/CaptureEngine.cpp src/app.rc)
target_link_libraries(CaptureTest PRIVATE VeilCommon user32 gdi32 ole32 windowscodecs)
target_link_options(CaptureTest PRIVATE /MANIFEST:NO)
add_executable(CaptureBroker src/broker/main.cpp src/broker/Benchmark.cpp src/broker/BenchmarkResources.cpp src/capture/CaptureEngine.cpp src/app.rc)
target_link_libraries(CaptureBroker PRIVATE VeilFrameReadback psapi pdh wtsapi32 user32 gdi32 ole32 windowscodecs)
target_link_options(CaptureBroker PRIVATE /MANIFEST:NO)
add_executable(SharedTextureProbe src/broker/Probe.cpp)
target_link_libraries(SharedTextureProbe PRIVATE VeilFrameReadback)
add_executable(AuthorizationEnvelopeProbe tests/AuthorizationEnvelopeProbe.cpp src/broker/AuthorizationEnvelope.cpp)
target_include_directories(AuthorizationEnvelopeProbe PRIVATE src)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT SecureViewer)
include(CTest)
if(BUILD_TESTING)
    add_executable(ProgressWatchdogProbe tests/ProgressWatchdogProbe.cpp)
    target_link_libraries(ProgressWatchdogProbe PRIVATE VeilCommon)
    add_test(NAME ProgressWatchdogProbe COMMAND ProgressWatchdogProbe)
    set_tests_properties(ProgressWatchdogProbe PROPERTIES LABELS "Headless" TIMEOUT 10)
    add_executable(ProcessIdentityProbe tests/ProcessIdentityProbe.cpp)
    target_link_libraries(ProcessIdentityProbe PRIVATE VeilBroker)
    add_test(NAME ProcessIdentityProbe COMMAND ProcessIdentityProbe)
    set_tests_properties(ProcessIdentityProbe PROPERTIES LABELS "Headless" TIMEOUT 15)
    add_executable(ViewerTextureProbe tests/ViewerTextureProbe.cpp src/window/SecureWindow.cpp
        src/graphics/D3D11Renderer.cpp src/security/DisplayProtection.cpp src/app.rc)
    target_link_libraries(ViewerTextureProbe PRIVATE VeilFrameReadback user32 d2d1 dwrite dwmapi)
    target_link_options(ViewerTextureProbe PRIVATE /MANIFEST:NO)
    add_test(NAME ViewerTextureProbe COMMAND ViewerTextureProbe)
    set_tests_properties(ViewerTextureProbe PROPERTIES LABELS "Interactive" RUN_SERIAL TRUE TIMEOUT 15)
    find_program(VEIL_POWERSHELL NAMES pwsh powershell REQUIRED)
    add_test(NAME WindowSmoke COMMAND "${VEIL_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_SOURCE_DIR}/tests/window-smoke.ps1" -BuildDir "$<TARGET_FILE_DIR:SecureViewer>")
    add_test(NAME CaptureSmoke COMMAND "${VEIL_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_SOURCE_DIR}/tests/capture-smoke.ps1" -BuildDir "$<TARGET_FILE_DIR:SecureViewer>"
        -ResultDir "${CMAKE_BINARY_DIR}/test-results")
    set_tests_properties(WindowSmoke CaptureSmoke PROPERTIES RUN_SERIAL TRUE TIMEOUT 75)
    add_test(NAME ZeroCopyStructure COMMAND "${VEIL_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_SOURCE_DIR}/tests/zero-copy-structure.ps1")
    add_test(NAME BrokerProtocol COMMAND CaptureBroker --protocol-tests)
    add_test(NAME AuthorizationEnvelopeProbe COMMAND AuthorizationEnvelopeProbe
        "${CMAKE_SOURCE_DIR}/tests/fixtures/authorization-envelope-v1.hex")
    add_test(NAME BrokerSmoke COMMAND "${VEIL_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_SOURCE_DIR}/tests/broker-smoke.ps1" -BuildDir "$<TARGET_FILE_DIR:CaptureBroker>"
        -ResultDir "${CMAKE_BINARY_DIR}/test-results" -ServiceExe "${CMAKE_SOURCE_DIR}/service/target/release/veil-secure-service.exe")
    set_tests_properties(ZeroCopyStructure BrokerProtocol AuthorizationEnvelopeProbe PROPERTIES LABELS "Headless")
    set_tests_properties(BrokerSmoke PROPERTIES RUN_SERIAL TRUE TIMEOUT 90)
    add_test(NAME BenchmarkAnalysis COMMAND "${VEIL_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_SOURCE_DIR}/tests/benchmark-analysis.ps1" -ResultDir "${CMAKE_BINARY_DIR}/test-results")
    set_tests_properties(BenchmarkAnalysis PROPERTIES LABELS "Headless" TIMEOUT 30)
    add_test(NAME BenchmarkSmoke COMMAND "${VEIL_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
        -File "${CMAKE_SOURCE_DIR}/tests/benchmark-smoke.ps1" -BuildDir "$<TARGET_FILE_DIR:CaptureBroker>"
        -ResultDir "${CMAKE_BINARY_DIR}/test-results" -ServiceExe "${CMAKE_SOURCE_DIR}/service/target/release/veil-secure-service.exe")
    set_tests_properties(BenchmarkSmoke PROPERTIES LABELS "Interactive" RUN_SERIAL TRUE TIMEOUT 30)
    add_test(NAME SharedTextureProbe COMMAND SharedTextureProbe)
    set_tests_properties(SharedTextureProbe PROPERTIES RUN_SERIAL TRUE TIMEOUT 15)
    set_tests_properties(WindowSmoke CaptureSmoke BrokerSmoke SharedTextureProbe PROPERTIES LABELS "Interactive")
endif()
